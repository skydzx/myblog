## 1. 端口与服务排查

```bash
# 当前监听端口
ss -tulnp | grep LISTEN
# 或
netstat -tulnp | grep LISTEN

# 所有运行中 / 开机自启的服务
systemctl list-units --type=service --state=running
systemctl list-unit-files --type=service --state=enabled
```

## 2. 关闭不必要的服务

```bash
SERVICES="telnet.socket rsh.socket rlogin.socket rexec.socket
          rpcbind avahi-daemon cups bluetooth nfs-server
          vsftpd xinetd tftp-server ypbind ypserv
          postfix sendmail"   # 若不需要邮件

for svc in $SERVICES; do
    if systemctl is-active --quiet "$svc" 2>/dev/null; then
        echo "🔴 关闭: $svc"
        systemctl disable --now "$svc"
    fi
done
```

## 3. 检查不安全服务

```bash
# Telnet（明文）
systemctl status telnet.socket 2>/dev/null
# FTP
systemctl status vsftpd ftpd 2>/dev/null
# R 系列（rsh/rlogin/rexec）
rpm -qa | grep rsh-server
# NFS（若无需求）
showmount -e localhost 2>/dev/null

# xinetd 子服务
systemctl status xinetd 2>/dev/null
for f in /etc/xinetd.d/*; do
    [[ -f "$f" ]] && echo "--- $f ---" && grep -E "disable|service" "$f"
done
```

## 4. 进程查看与监控

```bash
ps aux                    # 所有进程
ps -ef                    # System V 风格（含 PPID）
ps auxf                   # 进程树

# 筛选
ps aux | grep nginx
ps -u www-data            # 按用户
ps -p 1234               # 按 PID
ps --ppid 1              # 子进程

# 实时监控
top / htop / atop
watch -n 1 "ps aux | grep suspicious_name"

# 进程详细信息
cat /proc/<PID>/cmdline | tr '\0' ' '    # 完整启动命令
cat /proc/<PID>/status                   # 状态/权限
ls -la /proc/<PID>/exe                   # 可执行文件路径
ls -la /proc/<PID>/fd                    # 打开的文件描述符
readlink /proc/<PID>/cwd                 # 工作目录
```

## 5. 可疑进程排查（后门 / 内存马）

```bash
# 1. 已删除但仍运行的进程（可能是内存马/后门）
ls -la /proc/*/exe 2>/dev/null | grep deleted

# 2. 从 /tmp /var/tmp /dev/shm 运行的进程
ps -eo pid,comm,args | grep -E "^/tmp/|^/var/tmp/|^/dev/shm/"

# 3. 高 CPU / 内存占用
ps aux --sort=-%cpu | head -20
ps aux --sort=-%mem | head -20

# 4. 以 root 运行的异常进程
ps -eo pid,user,args | grep root | grep -v "$$" | grep -vE "systemd|sshd|cron|audit"

# 5. 隐藏进程（对比 /proc 与 ps）
diff <(ls /proc | grep -E '^[0-9]+$' | sort -n) \
     <(ps -eo pid --no-headers | sort -n) | grep "^[<>]"
```

## 6. 安全终止进程

```bash
# 优雅终止（SIGTERM），确认后再强制
kill <PID>
sleep 3
ps -p <PID> > /dev/null 2>&1 && echo "仍在运行" || echo "已终止"
kill -9 <PID>            # 未退出才强制

# 按名称 / 用户 / 终端
killall nginx
pkill -f "python.*http"
pkill -u <username>      # 踢人下线
pkill -t pts/0
```

> ⚠️ 不要随意 kill PID 1（systemd）；kill -9 不释放文件锁/socket，优先用 kill(15)；杀僵尸进程要先杀其父进程。

## 7. 端口查看与关闭

```bash
# 查看监听端口 / 特定端口 / 特定进程连接
ss -tulnp
ss -tulnp | grep :3306
lsof -i :22
ss -tulnp | grep <PID>

# 对比基线（当前端口 vs 预期端口）
ss -tlnp | awk '{print $4}' | grep -oE '[0-9]+$' | sort -n > /tmp/current_ports.txt
echo -e "22\n80\n443" > /tmp/expected_ports.txt
diff /tmp/expected_ports.txt /tmp/current_ports.txt

# 关闭端口三方法：
systemctl stop <service> && systemctl disable <service>   # ① 停服务
firewall-cmd --permanent --remove-port=3306/tcp; firewall-cmd --reload   # ② 防火墙
# 或 iptables -A INPUT -p tcp --dport 3306 -j DROP
# ③ 改应用绑定 localhost（如 MySQL: bind-address=127.0.0.1）
```

## 8. 进程资源限制（防 DoS / fork 炸弹）

```bash
cat >> /etc/security/limits.conf << 'EOF'
*    hard    nproc      1024
*    hard    nofile     4096
*    hard    core       0
*    hard    maxlogins  5
root hard    nproc      65535
root hard    nofile     65535
EOF

# systemd 服务级限制
# /etc/systemd/system/<service>.service.d/override.conf
[Service]
LimitNPROC=512
LimitNOFILE=1024
MemoryLimit=512M
```
